        In .py File
        from odoo import fields, models


        class FleetVehicle(models.Model):
            _inherit = "fleet.vehicle"

            latitude_in = fields.Float('Geo Latitude', digits=(16, 5))
            longitude_in = fields.Float('Geo Longitude', digits=(16, 5))
            latitude_out = fields.Float('Geo Latitude', digits=(16, 5))
            longitude_out = fields.Float('Geo Longitude', digits=(16, 5))

            def sign_in(self, kwargs=None):
                """
                :param kwargs: {'lat': FLOAT, 'lng': FLOAT, 'mode': 'readonly | edit'}
                """
                self.latitude_in = kwargs.get('lat')
                self.longitude_in = kwargs.get('lng')

            def sign_out(self, kwargs=None):
                """
                :param kwargs: {'lat': FLOAT, 'lng': FLOAT, 'mode': 'readonly | edit'}
                """
                self.latitude_out = kwargs.get('lat')
                self.longitude_out = kwargs.get('lng')
